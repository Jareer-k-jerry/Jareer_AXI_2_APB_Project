# ==============================================================================
#             AXI4-to-APB Bridge Verification - VCS Run Commands Cheat Sheet
# ==============================================================================
# All commands should be run from inside the 'sim/' directory:
#   cd sim
# ==============================================================================

# ------------------------------------------------------------------------------
# 1. RANDOMIZED TESTS (100 Iterations / Customizable via N=<count>)
# ------------------------------------------------------------------------------

# 100 Random Full Traffic Stress Test (Read + Write mix, random wait states & delays):
make run TEST=axi2apb_random_100_trans_test

# Random Single Write Test (100 randomized address/data writes):
make run TEST=axi2apb_random_single_write_test

# Random Single Read Test (100 randomized address reads):
make run TEST=axi2apb_random_single_read_test

# Random Nonzero APB Wait States Test (100 randomized wait states):
make run TEST=axi2apb_random_wait_state_test

# Random Write with BREADY Delays (100 writes with randomized backpressure):
make run TEST=axi2apb_random_bready_delay_test

# Random Read with RREADY Delays (100 reads with randomized backpressure):
make run TEST=axi2apb_random_rready_delay_test

# Random Simultaneous Read & Write Priority Test:
make run TEST=axi2apb_random_simultaneous_rw_test

# Random Back-to-Back Reads Test:
make run TEST=axi2apb_random_b2b_reads_test

# Random Back-to-Back Writes Test:
make run TEST=axi2apb_random_b2b_writes_test

# Random Mixed Read/Write Traffic Test:
make run TEST=axi2apb_random_mixed_traffic_test

# Random Diverse IDs & USER Sidebands Test:
make run TEST=axi2apb_random_diff_ids_users_test

# Random APB PSLVERR Error Injection Test:
make run TEST=axi2apb_random_pslverr_test

# Random Reset Smoke Test:
make run TEST=axi2apb_random_smoke_test

# --- HOW TO RUN WITH DIFFERENT ITERATION COUNTS (e.g. 50 or 200) ---
# make run TEST=axi2apb_random_bready_delay_test N=50
# make run TEST=axi2apb_random_bready_delay_test N=200


# ------------------------------------------------------------------------------
# 2. DIRECTED TESTS (Targeted Corner Cases)
# ------------------------------------------------------------------------------

# Reset / Smoke Test:
make run TEST=axi2apb_smoke_test

# Single Write Test (APB immediate ready):
make run TEST=axi2apb_single_write_test

# Single Read Test (APB immediate ready):
make run TEST=axi2apb_single_read_test

# Wait States Test (Write & Read with wait cycles):
make run TEST=axi2apb_wait_state_test

# Delayed RREADY Test:
make run TEST=axi2apb_rready_delay_test

# Delayed BREADY Test:
make run TEST=axi2apb_bready_delay_test

# Simultaneous Read & Write Priority Test (Checks READ services before WRITE):
make run TEST=axi2apb_simultaneous_rw_test

# Back-to-Back Reads Test (Pipeline check):
make run TEST=axi2apb_b2b_reads_test

# Back-to-Back Writes Test (FIFO buffer check):
make run TEST=axi2apb_b2b_writes_test

# Mixed Traffic Test (Alternating Reads & Writes):
make run TEST=axi2apb_mixed_traffic_test

# Different IDs & USER sidebands Test:
make run TEST=axi2apb_diff_ids_users_test

# APB PSLVERR Stimulus & Behavior Test:
make run TEST=axi2apb_pslverr_test


# ------------------------------------------------------------------------------
# 3. URG HTML COVERAGE REPORT COMMANDS
# ------------------------------------------------------------------------------

# Generate/Regenerate URG HTML report folder (urgReport/):
make report TEST=axi2apb_random_100_trans_test

# Open HTML Coverage Dashboard in browser:
make code_covg_gui
# Or directly:
firefox ./urgReport/dashboard.html &

# Open Functional Coverage Groups in browser:
make func_covg_gui
# Or directly:
firefox ./urgReport/groups.html &


# ------------------------------------------------------------------------------
# 4. WAVEFORM & DEBUGGING
# ------------------------------------------------------------------------------

# Open Waveforms in Synopsys DVE GUI:
make gui TEST=axi2apb_random_bready_delay_test


# ------------------------------------------------------------------------------
# 5. REGRESSIONS (Runs All Tests & Generates Merged Coverage Report)
# ------------------------------------------------------------------------------

# Run all Directed Tests regression:
make regression_directed

# Run all Randomized Tests regression:
make regression_random

# Run complete test suite:
make regression_all


# ------------------------------------------------------------------------------
# 6. CLEANING
# ------------------------------------------------------------------------------

# Clean one test build:
make clean TEST=axi2apb_random_bready_delay_test

# Clean all test builds and coverage reports:
make clean_all
